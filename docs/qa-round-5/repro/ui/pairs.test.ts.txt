import { it } from 'vitest'
import { writeFileSync } from 'node:fs'
import * as fs from 'node:fs'
import * as path from 'node:path'

const ROOT = path.join(process.cwd(), 'src/lib/i18n')
const files = [...fs.readdirSync(ROOT).filter(f => f.endsWith('.ts') && !f.includes('test')).map(f => path.join(ROOT, f)),
  ...fs.readdirSync(path.join(ROOT, 'budgetPage')).filter(f => f.endsWith('.ts') && !f.includes('test')).map(f => path.join(ROOT, 'budgetPage', f))]
const arabic = /[؀-ۿ]/
const ph = (s: string) => (s.match(/\{[^}]+\}|%[sd]|\$\{[^}]+\}/g) ?? []).sort().join('|')
function leaves(node: unknown, p: string, out: Map<string, unknown>) {
  if (typeof node === 'string' || typeof node === 'function') { out.set(p, node); return }
  if (Array.isArray(node)) { node.forEach((v, i) => leaves(v, `${p}[${i}]`, out)); return }
  if (node && typeof node === 'object') for (const [k, v] of Object.entries(node)) leaves(v, p ? `${p}.${k}` : k, out)
}
const lines: string[] = []
function cmp(name: string, E: any, A: any) {
  const e = new Map<string, unknown>(), a = new Map<string, unknown>()
  leaves(E, '', e); leaves(A, '', a)
  lines.push(`## ${name} en=${e.size} ar=${a.size}`)
  for (const k of e.keys()) if (!a.has(k)) lines.push(`MISSING_AR ${name} ${k}`)
  for (const k of a.keys()) if (!e.has(k)) lines.push(`MISSING_EN ${name} ${k}`)
  for (const [k, v] of e) {
    const x = a.get(k)
    if (x === undefined) continue
    if (typeof v !== typeof x) { lines.push(`TYPE ${name} ${k}`); continue }
    if (typeof v === 'string') {
      const s = x as string
      if (s.trim() === '') lines.push(`EMPTY_AR ${name} ${k}`)
      else if (s === v && /[A-Za-z]{3,}/.test(v)) lines.push(`SAME ${name} ${k} = ${JSON.stringify(v).slice(0, 60)}`)
      else if (!arabic.test(s) && /[A-Za-z]{4,}/.test(s)) lines.push(`NOARABIC ${name} ${k} = ${JSON.stringify(s).slice(0, 60)}`)
      if (ph(v) !== ph(s)) lines.push(`PLACEHOLDER ${name} ${k}`)
    } else {
      const f = v as Function, g = x as Function
      if (f.length !== g.length) lines.push(`ARITY ${name} ${k} en=${f.length} ar=${g.length}`)
      for (const args of [[1, 2, 3, 4], ['Rent', 'Food', 'Z', 'W'], [0, 0, 0, 0], [2, 2, 2, 2], [3, 3, 3, 3], [11, 11, 11, 11], [100, 'a', 'b', 'c']]) {
        try {
          const t = String(g(...args)), r = String(f(...args))
          if (t.trim() === '') { lines.push(`FN_EMPTY_AR ${name} ${k} ${JSON.stringify(args)}`); break }
          if (/undefined|NaN|\[object/.test(t) && !/undefined|NaN|\[object/.test(r)) { lines.push(`FN_BAD_AR ${name} ${k} ${JSON.stringify(args)} -> ${t.slice(0, 70)}`); break }
        } catch { }
      }
    }
  }
}
it('pairs', async () => {
  for (const file of files) {
    const mod: Record<string, any> = await import(file)
    const nm = path.basename(file)
    for (const key of Object.keys(mod)) {
      const m = key.match(/^(.*)En$/)
      if (m && mod[m[1] + 'Ar'] && typeof mod[key] === 'object') cmp(`${nm}:${m[1]}`, mod[key], mod[m[1] + 'Ar'])
      else if (typeof mod[key] === 'object' && mod[key] && mod[key].en && mod[key].ar && Object.keys(mod[key]).length === 2) cmp(`${nm}:${key}`, mod[key].en, mod[key].ar)
    }
  }
  writeFileSync(process.env.OUT!, lines.join('\n'))
})
