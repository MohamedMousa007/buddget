// Copy to src/__qa5__/dbl/linkIds.test.ts and run: npx vitest run src/__qa5__/dbl
// Observed: link_ids = ['answered-car'] reaches the uuid[] column (fails). Live logs: POST /rest/v1/budget_categories 400 x6/24h,
// postgres ERROR 'invalid input syntax for type uuid: "answered-car"'.
import { describe, it, expect } from 'vitest'
import { budgetCategoryToRow } from '@/lib/supabase/remote/mappers/budgetPlanMapper'

const UUID = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i

describe('DBL budget_categories.link_ids is uuid[]', () => {
  it('a studio fact-id link (engine.ts:172 linkIds:[d.id]) must not reach the uuid column', () => {
    const row = budgetCategoryToRow(
      { id: '11111111-1111-4111-8111-111111111111', name: 'Car loan', icon: 'landmark', amount: 100, block: 'debts', linkType: 'debts', linkIds: ['answered-car'] } as never,
      '22222222-2222-4222-8222-222222222222', '33333333-3333-4333-8333-333333333333', 0, 'EGP'
    )
    expect((row.link_ids as string[]).every((x) => UUID.test(x))).toBe(true)
  })
})
