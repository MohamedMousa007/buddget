import { describe, it, expect } from 'vitest'
import { getMonthRange, projectedIncomeForMonth, calculateMonthlyIncome } from '@/lib/utils/calculations'
import { paydayDatesForWindow } from '@/lib/utils/paydaySchedule'
import { src, key } from './_h'

describe('payday windows', () => {
  it('MON-1 monthly payday: one payday in every cycle (sd 1..28, pd 1..31, 2025-2027)', () => {
    const bad: string[] = []
    for (let sd = 1; sd <= 28; sd++) for (let pd = 1; pd <= 31; pd++) {
      const s = src({ dayOfMonth: pd, paydayDays: [pd] })
      for (let y = 2025; y <= 2027; y++) for (let m = 0; m < 12; m++) {
        const { start, end } = getMonthRange(key(y, m), sd)
        const n = paydayDatesForWindow(s, start, end).length
        if (n !== 1) bad.push(`sd=${sd} pd=${pd} ${key(y, m)} n=${n}`)
      }
    }
    expect(bad.slice(0, 8)).toEqual([])
  })
  it('MON-2 legacy weekly: projected vs monthly', () => {
    const s = src({ recurringFrequency: 'weekly', paydayDays: undefined, dayOfMonth: 3, amount: 100 })
    const diffs: string[] = []
    for (let m = 0; m < 12; m++) {
      const a = projectedIncomeForMonth([s], 'EGP', {}, key(2026, m), 1)
      const b = calculateMonthlyIncome([s], 'EGP', {}, key(2026, m), 1)
      if (a !== b) diffs.push(`${key(2026, m)} projected=${a} monthly=${b}`)
    }
    expect(diffs).toEqual([])
  })
})
