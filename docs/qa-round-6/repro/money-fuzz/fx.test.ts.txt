import { describe, it, expect } from 'vitest'
import { tryConvertCurrency, tryConvertWithGold, sumConverted } from '@/lib/utils/currency'
import { DEFAULT_MARKET_RATES } from '@/lib/store/defaultFinanceData'
import { goldGramsToMoney, moneyToGoldGrams } from '@/lib/utils/calculations'
import { formatQuantity } from '@/lib/utils/formatters'
import { rng } from './prng'

const FIAT = ['EGP', 'USD', 'EUR', 'GBP', 'AED', 'SAR', 'KWD', 'QAR', 'BHD', 'OMR']
// Same construction as src/app/api/rates/route.ts: USD_X + X_USD + every cross pair from one USD-based table
const USD_PER: Record<string, number> = { USD: 1, EGP: 53.1, EUR: 0.86, GBP: 0.75, AED: 3.6725, SAR: 3.75, KWD: 0.307, QAR: 3.64, BHD: 0.376, OMR: 0.385 }
const live = (() => { const t: Record<string, number> = {}; for (const a of FIAT) for (const b of FIAT) if (a !== b) t[`${a}_${b}`] = USD_PER[b] / USD_PER[a]; return t })()

describe('FX', () => {
  it('live-shaped table: A->B->A round trip drift < 1e-9 relative, all pairs, random amounts (incl. 0.005, 1e12)', () => {
    const r = rng(31337); const bad: string[] = []; let cases = 0
    for (let i = 0; i < 40000; i++) {
      const a = r.pick(FIAT), b = r.pick(FIAT); if (a === b) continue
      const x = r.pick([0.005, 0.01, 1e12, r.next() * 1000, r.next() * 1e7, 1e-7])
      const there = tryConvertCurrency(x, a, b, live); const back = there === null ? null : tryConvertCurrency(there, b, a, live)
      cases++
      if (there === null || back === null) { bad.push(`NULL ${a}->${b}`); continue }
      if (Math.abs(back - x) > 1e-9 * x) bad.push(`DRIFT ${a}->${b} x=${x} back=${back}`)
    }
    console.log('fx live cases', cases, 'bad', bad.length, bad.slice(0, 4)); expect(bad).toEqual([])
  })
  it('built-in DEFAULT_MARKET_RATES (used until the first rates fetch): round trip drift per pair', () => {
    const worst: string[] = []
    for (const a of FIAT) for (const b of FIAT) { if (a === b) continue
      const there = tryConvertCurrency(100000, a, b, DEFAULT_MARKET_RATES); if (there === null) continue
      const back = tryConvertCurrency(there, b, a, DEFAULT_MARKET_RATES)!
      if (Math.abs(back - 100000) > 1) worst.push(`${a}->${b}->${a}: 100000 -> ${back.toFixed(2)}`)
    }
    console.log('DEFAULT drift pairs >1 unit per 100k:', worst.length, JSON.stringify(worst.slice(0, 8)))
  })
  it('conversion is multiplicative-consistent: A->C == A->B->C (live table), 3-decimal currencies keep KWD/BHD/OMR precision', () => {
    const r = rng(5); const bad: string[] = []
    for (let i = 0; i < 20000; i++) { const [a, b, c] = [r.pick(FIAT), r.pick(FIAT), r.pick(FIAT)]; const x = r.next() * 5000
      const d = tryConvertCurrency(x, a, c, live)!, v = tryConvertCurrency(tryConvertCurrency(x, a, b, live)!, b, c, live)!
      if (Math.abs(d - v) > 1e-9 * Math.max(1, d)) bad.push(`${a}>${b}>${c} ${d} vs ${v}`) }
    console.log('fx triangle bad', bad.length, bad.slice(0, 3)); expect(bad).toEqual([])
  })
  it('XAU grams to 4dp: grams->money->grams drift, formatQuantity(gold) keeps 3 dp (stored precision beyond display)', () => {
    const r = rng(8); const bad: string[] = []
    for (let i = 0; i < 20000; i++) { const g = Math.round(r.next() * 1e6) / 1e4; const k = r.pick([24, 22, 21, 18] as const)
      const m = goldGramsToMoney(g, 4000 + r.next() * 500, k); const back = moneyToGoldGrams(m, 4000, k)
      const price = 4000 + r.next() * 500
      const rt = moneyToGoldGrams(goldGramsToMoney(g, price, k), price, k)
      if (Math.abs(rt - g) > 1e-9 * Math.max(1, g)) bad.push(`g=${g} k=${k} rt=${rt}`)
      void back }
    const q = formatQuantity(0.00049, 'gold'), q2 = formatQuantity(1.23449, 'gold')
    console.log('gold rt bad', bad.length, bad.slice(0, 3), 'fmt', q, q2); expect(bad).toEqual([])
  })
  it('fail-closed: missing/NaN/0/negative rates never produce NaN or a 1:1 conversion', () => {
    const r = rng(11); const bad: string[] = []
    for (let i = 0; i < 20000; i++) { const t: Record<string, number> = {}
      for (const a of FIAT) for (const b of FIAT) if (a !== b && r.chance(0.15)) t[`${a}_${b}`] = r.pick([NaN, 0, -3, Infinity, 1e-300, 5, 0.2])
      const a = r.pick(FIAT), b = r.pick(FIAT); const x = r.next() * 100 + 0.01
      const v = tryConvertCurrency(x, a, b, t)
      if (v !== null && (!Number.isFinite(v) || v < 0)) bad.push(`${a}>${b} ${JSON.stringify(t[`${a}_${b}`])} -> ${v}`) }
    console.log('fx failclosed bad', bad.length, JSON.stringify(bad.slice(0, 11))); expect(bad).toEqual([])
  })
  it('sumConverted lower bound flag', () => { expect(sumConverted([1, 2], (x) => (x === 1 ? null : x))).toEqual({ total: 2, incomplete: true }); void tryConvertWithGold })
})
