rules:
  - id: sst-math-random-secret
    languages: [typescript]
    severity: WARNING
    message: "Math.random() is not a CSPRNG; do not use it for tokens, secrets, nonces, OTP codes or ids an attacker must not guess"
    patterns:
      - pattern-either:
          - pattern: const $N = $E
          - pattern: let $N = $E
          - pattern: $N = $E
      - metavariable-pattern:
          metavariable: $E
          pattern: <... Math.random() ...>
      - metavariable-regex:
          metavariable: $N
          regex: (?i).*(token|secret|nonce|otp|code|password|salt|session|csrf|hash|key).*
