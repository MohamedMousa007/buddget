import { renderLite } from './liteReact'
import { useFinanceStore } from '@/lib/store/useFinanceStore'
import { useSettingsStore } from '@/lib/store/useSettingsStore'
import { useAddExpenseSheet } from '@/hooks/useAddExpenseSheet'
import { useNewGoalForm } from '@/hooks/useNewGoalForm'
import { canonicalAmount, parseLocalizedAmount } from '@/lib/ui/amountParse'

const pm = { id: 'pm1', name: 'Cash', type: 'cash', isDefault: true } as never

beforeEach(() => {
  const s = useFinanceStore.getState()
  useFinanceStore.setState({
    expenses: [], goals: [], savingsAccounts: [], paymentMethods: [pm],
    settings: { ...s.settings, baseCurrency: 'EGP', showSecondaryCurrency: false, showAllCurrenciesInForms: true },
  })
  useSettingsStore.setState({ activeModal: 'addExpense', expensePrefill: null } as never)
})

describe('add expense', () => {
  it('amount typing cannot produce 0.005 / 1e21 / Arabic digits', () => {
    expect(canonicalAmount('0.005', 2)).toBe('0.00')
    expect(canonicalAmount('1e21', 2)).toBe('')
    expect(canonicalAmount('١٢٫٥٠', 2)).toBe('12.50')
    expect(canonicalAmount('EGP 1,250.50', 2)).toBe('1250.50')
    expect(canonicalAmount('1,250', 3)).toBe('1.250')
    expect(canonicalAmount('9'.repeat(30), 2).length).toBe(12)
  })
  it('FRM-currency: 3-decimal amount survives a switch to a 2-decimal currency', () => {
    const h = renderLite(() => useAddExpenseSheet())
    h.act(() => { h.result.current.setCurrency('KWD' as never); h.result.current.setAmount(canonicalAmount('1.234', 3)); h.result.current.setDescription('x') })
    h.act(() => h.result.current.setCurrency('EGP' as never))
    h.act(() => h.result.current.handleSubmit())
    const e = useFinanceStore.getState().expenses
    expect(e.length).toBe(1)
    console.log('saved', e[0].amount, e[0].currency)
    expect(e[0].amount).toBe(1.23)
  })
  it('FRM-prefill: AI/SMS prefill with grouped or Arabic amount', () => {
    useSettingsStore.setState({ expensePrefill: { amount: '1,250.50', description: 'x', currency: 'EGP' } } as never)
    const h = renderLite(() => useAddExpenseSheet())
    console.log('prefilled amount state', JSON.stringify(h.result.current.amount), h.result.current.isOpen, JSON.stringify(useSettingsStore.getState().expensePrefill))
    h.act(() => h.result.current.handleSubmit())
    console.log('rows', useFinanceStore.getState().expenses.map((e) => e.amount))
    expect(useFinanceStore.getState().expenses[0]?.amount).toBe(1250.5)
  })
  it('FRM-prefill-arabic', () => {
    useSettingsStore.setState({ expensePrefill: { amount: '٥٠', description: 'x', currency: 'EGP' } } as never)
    const h = renderLite(() => useAddExpenseSheet())
    h.act(() => h.result.current.handleSubmit())
    expect(useFinanceStore.getState().expenses[0]?.amount).toBe(50)
  })
  it('whitespace description passes hook gate', () => {
    const h = renderLite(() => useAddExpenseSheet())
    h.act(() => { h.result.current.setAmount('5'); h.result.current.setDescription('   ') })
    h.act(() => h.result.current.handleSubmit())
    console.log('whitespace rows', useFinanceStore.getState().expenses.length, JSON.stringify(useFinanceStore.getState().expenses[0]?.description))
  })
  it('sequential double submit is idempotent (state resets)', () => {
    const h = renderLite(() => useAddExpenseSheet())
    h.act(() => { h.result.current.setAmount('5'); h.result.current.setDescription('x') })
    h.act(() => h.result.current.handleSubmit())
    h.act(() => h.result.current.handleSubmit())
    expect(useFinanceStore.getState().expenses.length).toBe(1)
  })
  it('FRM-exit: the frozen exiting-sheet handler (what AnimatePresence keeps clickable) saves again', () => {
    const h = renderLite(() => useAddExpenseSheet())
    h.act(() => { h.result.current.setAmount('5'); h.result.current.setDescription('x') })
    const frozen = h.result.current.handleSubmit
    h.act(() => frozen())
    h.act(() => frozen())
    expect(useFinanceStore.getState().expenses.length).toBe(1)
  })
})

describe('new goal', () => {
  it('FRM-goal: double submit creates two goals, two pockets, two opening deposits', () => {
    const done = vi.fn()
    const h = renderLite(() => useNewGoalForm({ category: 'custom' } as never, done))
    h.act(() => { h.result.current.setName('Car'); h.result.current.setTargetAmount('1000'); h.result.current.setAlreadySaved('300') })
    const submit = h.result.current.submit
    h.act(() => submit())
    h.act(() => h.result.current.submit())
    const st = useFinanceStore.getState()
    console.log('goals', st.goals.length, 'pockets', st.savingsAccounts.length, 'done calls', done.mock.calls.length)
    expect(st.goals.length).toBe(1)
  })
})

import { useEditExpenseForm } from '@/hooks/useEditExpenseForm'
import { useAddSubscriptionForm } from '@/hooks/useAddSubscriptionForm'

describe('gates and resets (clean checks)', () => {
  it('CLEAN add-expense gate: 0, empty, negative, blank description do not write', () => {
    const h = renderLite(() => useAddExpenseSheet())
    for (const [a, d] of [['0', 'x'], ['', 'x'], ['-5', 'x'], ['5', '']] as const) {
      h.act(() => { h.result.current.setAmount(a); h.result.current.setDescription(d) })
      h.act(() => h.result.current.handleSubmit())
    }
    expect(useFinanceStore.getState().expenses.length).toBe(0)
  })
  it('CLEAN add-expense reopen: close resets every field', () => {
    const h = renderLite(() => useAddExpenseSheet())
    h.act(() => { h.result.current.setAmount('5'); h.result.current.setDescription('x'); h.result.current.setNotes('n'); h.result.current.setCurrency('USD' as never) })
    h.act(() => h.result.current.handleClose())
    const r = h.result.current
    expect({ a: r.amount, d: r.description, n: r.notes, c: r.currency }).toEqual({ a: '', d: '', n: '', c: 'EGP' })
  })
  for (const [amount, want] of [['1e+21', 'a sane amount'], ['12.3456', 12.35]] as const) {
    it(`FRM-prefill-bounds ${amount}: an agent/AI prefill reaches the ledger unsanitised`, () => {
      useSettingsStore.setState({ activeModal: 'addExpense', expensePrefill: { amount, description: 'x', currency: 'EGP' } } as never)
      const h = renderLite(() => useAddExpenseSheet())
      h.act(() => h.result.current.handleSubmit())
      expect(useFinanceStore.getState().expenses[0]?.amount).toBe(want)
    })
  }
  it('CLEAN edit-expense: no FX rate -> error, nothing written', () => {
    const e = { id: 'e1', date: '2026-10-01', description: 'H', category: 'Food', amount: 20, currency: 'EGP', amountInBaseCurrency: 20, paymentMethodId: 'pm1', isRecurring: false, createdAt: '', updatedAt: '' } as never
    useFinanceStore.setState({ expenses: [e], exchangeRates: {} as never })
    const h = renderLite(() => useEditExpenseForm(e, vi.fn()))
    h.act(() => h.result.current.setCurrency('JOD' as never))
    h.act(() => h.result.current.handleSubmit())
    expect(h.result.current.submitError).not.toBe('')
    expect(useFinanceStore.getState().expenses[0].currency).toBe('EGP')
  })
  it('CLEAN edit-expense: row deleted meanwhile -> save is a no-op, no throw', () => {
    const e = { id: 'e1', date: '2026-10-01', description: 'H', category: 'Food', amount: 20, currency: 'EGP', amountInBaseCurrency: 20, paymentMethodId: 'pm1', isRecurring: false, createdAt: '', updatedAt: '' } as never
    useFinanceStore.setState({ expenses: [] })
    const h = renderLite(() => useEditExpenseForm(e, vi.fn()))
    h.act(() => h.result.current.setNotes('x'))
    expect(() => h.act(() => h.result.current.handleSubmit())).not.toThrow()
    expect(useFinanceStore.getState().expenses.length).toBe(0)
  })
  it('CLEAN subscription gate: 0 amount and share > amount blocked', () => {
    const h = renderLite(() => useAddSubscriptionForm(null, () => {}, null))
    h.act(() => h.result.current.patch({ name: 'X', amount: 0 }))
    expect(h.result.current.canSubmit).toBe(false)
    h.act(() => h.result.current.patch({ amount: 10, sharedWith: ['a'], myShare: 11 }))
    expect(h.result.current.canSubmit).toBe(false)
    h.act(() => h.result.current.patch({ myShare: 5 }))
    expect(h.result.current.canSubmit).toBe(true)
  })
  it('FRM-sub-double: a second Save on the same open sheet adds a second subscription', () => {
    useFinanceStore.setState({ subscriptions: [] })
    const onClose = vi.fn()
    const h = renderLite(() => useAddSubscriptionForm(null, onClose, null))
    h.act(() => h.result.current.patch({ name: 'X', amount: 10 }))
    h.act(() => h.result.current.submit())
    h.act(() => h.result.current.submit())
    expect(useFinanceStore.getState().subscriptions.length).toBe(1)
  })
})

describe('parse table + goal gate (clean checks)', () => {
  it('CLEAN parseLocalizedAmount', () => {
    const p = parseLocalizedAmount
    expect(p('١٬٢٥٠٫٥٠', 'EGP')).toBe(1250.5)
    expect(p('1,250', 'EGP')).toBe(1250)
    expect(p('1,250', 'KWD')).toBe(1.25)
    expect(p('1.5.5', 'EGP')).toBeNaN()
    expect(p('1e3', 'EGP')).toBeNaN()
    expect(p('0.005', 'EGP')).toBe(0.01)
    expect(p('', 'EGP')).toBeNaN()
    expect(p('007.50', 'EGP')).toBe(7.5)
  })
  it('CLEAN goal gate: zero/empty target cannot advance, default name when blank', () => {
    const h = renderLite(() => useNewGoalForm({ category: 'custom' } as never, vi.fn()))
    h.act(() => h.result.current.setName('Car'))
    expect(h.result.current.canNext2).toBe(false)
    h.act(() => h.result.current.setTargetAmount('0'))
    expect(h.result.current.canNext2).toBe(false)
    h.act(() => h.result.current.setTargetAmount('500'))
    expect(h.result.current.canNext2).toBe(true)
  })
})
