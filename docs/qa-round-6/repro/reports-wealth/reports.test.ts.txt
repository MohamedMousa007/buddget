const mulberry32 = (a: number) => () => { a |= 0; a = (a + 0x6d2b79f5) | 0; let t = Math.imul(a ^ (a >>> 15), 1 | a); t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t; return ((t ^ (t >>> 14)) >>> 0) / 4294967296 }
const RATES = { USD_EGP: 50, EGP_USD: 0.02, AED_EGP: 13.6, EGP_AED: 1 / 13.6 }
const mk = (o: Partial<Expense>): Expense => ({ id: Math.random().toString(36), date: '2026-11-03', description: 'x', category: 'Food', amount: 10, currency: 'EGP', amountInBaseCurrency: 10, paymentMethodId: 'pm1', isRecurring: false, ...o } as Expense)

describe('csv escaping', () => {
  const cases: [unknown, string][] = [
    ['=1+1', `"'=1+1"`], ['+cmd', `"'+cmd"`], ['-2+3', `"'-2+3"`], ['@SUM(A1)', `"'@SUM(A1)"`], ['\t=1', `"'\t=1"`],
    ['-5', '"-5"'], [-5, '"-5"'], ['مطعم', '"مطعم"'], ['a"b', '"a""b"'], ['a,b\nc', '"a,b\nc"'],
  ]
  for (const [v, exp] of cases) it(JSON.stringify(v), () => expect(escapeCsvField(v)).toBe(exp))
  it('leading space / newline / fullwidth before formula', () => {
    // Excel trims nothing: " =1+1" stays text; but "\n=1+1" and fullwidth are text too. Record what we emit.
    expect(escapeCsvField(' =1+1')).toBe('" =1+1"')
    expect(escapeCsvField('\n=1+1')).toBe('"\n=1+1"')
  })
  it('-1e5 string treated as number (plain), -1+1 text escaped', () => {
    expect(escapeCsvField('-1e5')).toBe('"-1e5"')
    expect(escapeCsvField('+1e5+cmd')).toBe(`"'+1e5+cmd"`)
  })
})

describe('refunds in report aggregates', () => {
  it('refunded/declined rows are counted as transactions in method breakdown and add 0-value categories', () => {
    const rows = reportSpendRows([mk({ amount: 100 }), mk({ amount: 50, refundKind: 'refunded', paymentMethodId: 'pm2', category: 'Shopping' }), mk({ amount: 70, refundKind: 'declined', paymentMethodId: 'pm2', category: 'Shopping' })])
    const mb = getPaymentMethodBreakdown(rows, [{ id: 'pm1', name: 'A' }, { id: 'pm2', name: 'B' }], 'EGP', RATES)
    const b = mb.find((m) => m.name === 'B')!
    console.log('breakdown', JSON.stringify(mb))
    // expected by a user: refunded/declined charges are not purchases -> count 0
    expect(b.count).toBe(0)
  })
  it('largest expense degenerates to a 0 row when all are refunded', () => {
    const rows = reportSpendRows([mk({ amount: 50, refundKind: 'refunded' })])
    expect(rows.map((e) => expenseAmountInBase(e, 'EGP', RATES))).toEqual([0])
    expect(rows.length).toBe(0) // expected: refunded row dropped from spend rows
  })
})

describe('report presets', () => {
  it('thisYear with monthStartDay>1 early in January shows the previous calendar year', () => {
    const now = new Date(2026, 0, 10)
    const c = reportCycles('thisYear', now, 15)
    console.log('thisYear Jan10 sd15', c[0], c[c.length - 1], c.length)
    const r = reportRange('thisYear', now, 15)
    // 'today' (Jan 10 2026) must lie inside the range
    expect(now >= r.start && now <= r.end).toBe(true)
    expect(c[c.length - 1].startsWith('2026') || r.end >= now).toBe(true)
  })
  it('every preset range contains today and tiles without gaps (fuzz, start day 1..31)', () => {
    const rnd = mulberry32(20261106)
    for (let i = 0; i < 4000; i++) {
      const now = new Date(2024 + Math.floor(rnd() * 4), Math.floor(rnd() * 12), 1 + Math.floor(rnd() * 28), Math.floor(rnd() * 24))
      const sd = 1 + Math.floor(rnd() * 31)
      for (const p of ['thisMonth', 'lastMonth', 'last3Months', 'last6Months', 'thisYear'] as const) {
        const keys = reportCycles(p, now, sd)
        const r = reportRange(p, now, sd)
        if (p !== 'lastMonth') expect(now >= r.start && now <= r.end, `${p} ${now.toISOString()} sd${sd}`).toBe(true)
        for (let k = 1; k < keys.length; k++) {
          const a = getMonthRange(keys[k - 1], sd).end, b = getMonthRange(keys[k], sd).start
          const nd = new Date(a.getFullYear(), a.getMonth(), a.getDate() + 1); expect(`${b.getFullYear()}-${b.getMonth()}-${b.getDate()}`).toBe(`${nd.getFullYear()}-${nd.getMonth()}-${nd.getDate()}`)
        }
      }
    }
  })
})

describe('differential: category totals vs naive', () => {
  it('sum(categoryData)=totalExpenses=sum(method totals); foreign currency; refunds; nonspend', () => {
    const rnd = mulberry32(77)
    const cats = ['Food', 'Rent', 'Transfer', 'Savings', 'Top up', 'Shopping', 'Remittance', 'Debt']
    const exps: Expense[] = []
    for (let i = 0; i < 20000; i++) {
      const cur = (['EGP', 'USD', 'AED'] as const)[Math.floor(rnd() * 3)]
      exps.push(mk({ id: 'e' + i, date: `2026-${String(1 + Math.floor(rnd() * 12)).padStart(2, '0')}-${String(1 + Math.floor(rnd() * 28)).padStart(2, '0')}`, category: cats[Math.floor(rnd() * cats.length)], amount: Math.round(rnd() * 1e5) / 100, currency: cur, paymentMethodId: 'pm' + Math.floor(rnd() * 3), refundKind: rnd() < 0.05 ? 'refunded' : undefined, spendExclusion: rnd() < 0.05 ? 'manual' : undefined }))
    }
    const range = { start: new Date(2026, 3, 1), end: new Date(2026, 5, 30, 23, 59, 59) }
    const inRange = filterExpensesByRange(exps, range)
    const naiveIn = exps.filter((e) => e.date >= '2026-04-01' && e.date <= '2026-06-30')
    expect(inRange.length).toBe(naiveIn.length)
    const rows = reportSpendRows(inRange)
    let naive = 0
    const rate = (c: string) => (c === 'EGP' ? 1 : c === 'USD' ? 50 : 13.6)
    for (const e of naiveIn) if (!['Savings', 'Transfer', 'Top up'].includes(e.category) && !e.spendExclusion && !e.refundKind) naive += e.amount * rate(e.currency)
    const total = rows.reduce((s, e) => s + expenseAmountInBase(e, 'EGP', RATES), 0)
    expect(Math.abs(total - naive)).toBeLessThan(1e-4 * Math.max(1, naive))
    const byCat: Record<string, number> = {}
    for (const e of rows) byCat[e.category] = (byCat[e.category] || 0) + expenseAmountInBase(e, 'EGP', RATES)
    expect(Math.abs(Object.values(byCat).reduce((a, b) => a + b, 0) - total)).toBeLessThan(1e-6 * total)
  })
})
