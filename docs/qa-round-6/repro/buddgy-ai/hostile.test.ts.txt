function walk(v: unknown, path: string, out: string[], seen = new WeakSet<object>(), depth = 0): void {
  if (depth > 6) return
  if (typeof v === 'number') {
    if (!Number.isFinite(v) || Math.abs(v) > 1e12) out.push(`${path}=${v}`)
    return
  }
  if (v && typeof v === 'object') {
    if (seen.has(v)) return
    seen.add(v)
    for (const [k, x] of Object.entries(v)) if (typeof x !== 'function') walk(x, `${path}.${k}`, out, seen, depth + 1)
  }
}

  for (const [k, v] of Object.entries(s)) if (typeof v !== 'function') walk(v, k, out)
  return out
}

describe('hostile args, every write tool', () => {
  const writes = [...TOOLS.values()].filter((t) => t.write)
  it('propose never throws and never mutates; apply never throws; stored numbers stay sane', () => {
    let n = 0
    for (const t of writes) {
      const name = t.spec.name
      const keys = Object.keys(t.spec.params.properties ?? {})
      const bases = [{}, BASE[name] ?? {}]
      for (const base of bases) {
        for (const k of keys) {
          for (const h of HOSTILE) {
            const args = { ...base, [k]: h } as Record<string, unknown>
            const before = snap()
            let r
            try {
              r = t.write!.propose(args, ctx)
            } catch (e) {
              report.push(`THROW propose ${name}.${k}=${label(h)}: ${(e as Error).message}`)
              continue
            }
            n++
            if (snap() !== before) report.push(`MUTATED-ON-PROPOSE ${name}.${k}=${label(h)}`)
            if (!r.ok) continue
            const pre = new Set(anomalies())
            try {
              const a = t.write!.apply(r.proposal, ctx)
              if (a.ok) {
                const post = anomalies().filter((x) => !pre.has(x))
                if (post.length) report.push(`BAD-NUMBER apply ${name}.${k}=${label(h)} -> ${post.slice(0, 3).join('; ')}`)
              }
            } catch (e) {
              report.push(`THROW apply ${name}.${k}=${label(h)}: ${(e as Error).message}`)
            }
            // reset store for next case
            useFinanceStore.setState({ expenses: [], debts: [], debtPayments: [], incomeEvents: [], incomeSources: [SALARY], paymentMethods: [{ id: PM, name: 'Cash', type: 'cash', currency: 'EGP', isDefault: true } as never] })
          }
        }
      }
    }
    report.push(`cases ${n}`)
    report.push(`pollution: ${JSON.stringify(({} as Record<string, unknown>).polluted)}`)
    writeFileSync('/private/tmp/claude-501/-Users-mohamedmousa-Desktop-Budget-Manager-buddget--claude-worktrees-app-qa-check-plan-e516fe/36154420-f3eb-42dd-95f4-d13552466269/scratchpad/hostile.txt', [...new Set(report)].join('\n'))
    expect(true).toBe(true)
  }, 120_000)
})
