function walk(v: unknown, path: string, out: string[], seen = new WeakSet<object>(), depth = 0): void {
  if (depth > 6) return
  if (typeof v === 'number') {
    if (!Number.isFinite(v) || Math.abs(v) > 1e12) out.push(`${path}=${v}`)
    return
  }
  if (v && typeof v === 'object') {
    if (seen.has(v)) return
    seen.add(v)
    for (const [k, x] of Object.entries(v)) if (typeof x !== 'function') walk(x, `${path}.${k}`, out, seen, depth + 1)
  }
}

  return out
}

function seed(): Record<string, string> {
  const run = (tool: string, args: Record<string, unknown>) => {
    const r = TOOLS.get(tool)!.write!.propose(args, ctx)
    if (!r.ok) throw new Error(tool + ' seed propose failed: ' + r.error)
    const a = TOOLS.get(tool)!.write!.apply(r.proposal, ctx)
    if (!a.ok) throw new Error(tool + ' seed apply failed: ' + a.error)
  }
  run('add_expense', { description: 'Lunch', amount: 100, category: 'Food', date: '2026-10-18' })
  run('add_income', BASE.add_income)
  run('add_debt', BASE.add_debt)
  run('add_subscription', BASE.add_subscription)
  run('add_goal', BASE.add_goal)
  run('add_savings_account', BASE.add_savings_account)
  run('add_savings_account', { ...BASE.add_savings_account, name: 'Second' })
  run('add_investment_holding', BASE.add_investment_holding)
  const s = useFinanceStore.getState()
  run('record_debt_payment', { debt: refOfId(s.debts[0].id), amount: 10 })
  const t = useFinanceStore.getState()
  return {
    expense: refOfId(t.expenses[0].id),
    income: refOfId(t.incomeSources[t.incomeSources.length - 1].id),
    pm: refOfId(t.paymentMethods[0].id),
    debt: refOfId(t.debts[0].id),
    sub: refOfId(t.subscriptions[0].id),
    goal: refOfId(t.goals[0].id),
    pocket: refOfId(t.savingsAccounts[0].id),
    pocket2: refOfId(t.savingsAccounts[1].id),
    asset: refOfId(t.investmentHoldings[0].id),
    payment: refOfId(t.debtPayments[0].id),
  }
}
