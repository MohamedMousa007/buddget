import { describe, it, expect } from 'vitest'
import { resolveCarryDestination } from '@/lib/savings/monthCarry'
import type { Goal, SavingsAccount } from '@/lib/store/types'

describe('SLG-1 create-goal-pocket never links', () => {
  it('second cycle after the pocket was created must not ask for another', () => {
    const goal = { id: 'g1', name: 'Car', status: 'active', category: 'purchase', priority: 1, targetDate: null, createdAt: '2026-01-01', linkedSavingsAccountIds: [], fundingMode: 'manual' } as unknown as Goal
    const first = resolveCarryDestination(null, [], [goal])
    expect(first.kind).toBe('create-goal-pocket')
    // what the route leaves behind: a pocket named after the goal, goal.linkedSavingsAccountIds unchanged
    const created = { id: 'p1', name: 'Car', category: 'savings', type: 'vault', currency: 'EGP', currentBalance: 100 } as unknown as SavingsAccount
    const second = resolveCarryDestination(null, [created], [goal])
    expect(second.kind).toBe('pocket')
  })
})
