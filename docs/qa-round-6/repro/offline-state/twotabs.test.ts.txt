import { afterEach, describe, expect, it, vi } from 'vitest'

describe('two tabs share localStorage', () => {
  afterEach(() => vi.unstubAllGlobals())
  it('tab B persisting after tab A overwrites A edits (no storage-event rehydrate)', async () => {
    const data = new Map<string, string>()
    const mkWin = () => { const bus = new EventTarget(); return { localStorage: { getItem: (k: string) => data.get(k) ?? null, setItem: (k: string, v: string) => void data.set(k, v), removeItem: (k: string) => void data.delete(k) }, addEventListener: bus.addEventListener.bind(bus), removeEventListener: bus.removeEventListener.bind(bus), dispatchEvent: bus.dispatchEvent.bind(bus) } }
    const hide = () => { const d = document as unknown as { visibilityState: string } & EventTarget; d.visibilityState = 'hidden'; d.dispatchEvent(new Event('visibilitychange')); d.visibilityState = 'visible' }
    const load = async () => { vi.resetModules(); vi.stubGlobal('window', mkWin()); vi.stubGlobal('document', Object.assign(new EventTarget(), { visibilityState: 'visible' })); const m = await import('@/lib/store/useFinanceStore'); return { m, hide: hide } }
    const A = await load(); const a = A.m.useFinanceStore
    const Bm = await load(); const b = Bm.m.useFinanceStore
    const exp = { id: '11111111-1111-4111-8111-111111111111', date: '2026-10-03', amount: 50, category: 'Food', description: 'x', currency: 'EGP' } as never
    // tab A edits (state lives in A's module) and persists
    ;(globalThis as { document: unknown }).document = undefined as never
    a.setState({ expenses: [exp] })
    // flush A's pending write by its own document listener
    await new Promise((r) => setTimeout(r, 700))
    const afterA = JSON.parse(data.get('buddget-storage') as string).state.expenses.length
    b.setState({ goldPricePerGram: 123 })
    await new Promise((r) => setTimeout(r, 700))
    const afterB = JSON.parse(data.get('buddget-storage') as string).state.expenses.length
    expect({ afterA, afterB }).toEqual({ afterA: 1, afterB: 1 })
  }, 20000)
})
