import { readFileSync } from 'node:fs'
import { describe, expect, it } from 'vitest'

const css = readFileSync('src/app/globals.css', 'utf8')
const d = css.indexOf('\n.dark {')
const th = { light: css.slice(0, d), dark: css.slice(d) }
type RGB = [number, number, number]
const hex = (h: string): RGB => [1, 3, 5].map((i) => parseInt(h.slice(i, i + 2), 16)) as RGB
const lin = (v: number) => ((v /= 255) <= 0.03928 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4)
const lum = (c: RGB) => 0.2126 * lin(c[0]) + 0.7152 * lin(c[1]) + 0.0722 * lin(c[2])
const ratio = (a: RGB, b: RGB) => (Math.max(lum(a), lum(b)) + 0.05) / (Math.min(lum(a), lum(b)) + 0.05)
const tok = (t: 'light' | 'dark', n: string) => hex(th[t].match(new RegExp(`${n}: (#[0-9A-Fa-f]{6})`))![1])
const WHITE: RGB = [255, 255, 255]

describe('text-white on solid semantic fills (buttonVariants success/destructive-solid, ActionToast, PromptBanner, SwipeToDelete, PullToRefresh)', () => {
  for (const theme of ['light', 'dark'] as const)
    for (const k of ['red', 'green', 'amber', 'red-hover', 'green-hover'])
      it(`${theme}: white on --color-brand-${k} >= 4.5`, () => {
        expect(ratio(WHITE, tok(theme, `--color-brand-${k}`))).toBeGreaterThanOrEqual(4.5)
      })
})

describe('muted/semantic text on the wells it is used on', () => {
  for (const theme of ['light', 'dark'] as const) {
    for (const bg of ['sunken', 'red-soft', 'green-soft', 'amber-soft', 'primary-soft'])
      it(`${theme}: text-muted on ${bg} >= 4.5`, () => {
        expect(ratio(tok(theme, '--color-brand-text-muted'), tok(theme, `--color-brand-${bg}`))).toBeGreaterThanOrEqual(4.5)
      })
    for (const fg of ['gold', 'primary', 'red', 'green', 'amber'])
      for (const bg of ['bg', 'elevated', 'sunken'])
        it(`${theme}: ${fg} text on ${bg} >= 4.5`, () => {
          expect(ratio(tok(theme, `--color-brand-${fg}`), tok(theme, `--color-brand-${bg}`))).toBeGreaterThanOrEqual(4.5)
        })
  }
})

describe('payment card face: white text on every user-selectable card colour (PaymentCardCarousel uses fixed text-white, no inkOn())', () => {
  const colours = readFileSync('src/lib/payment/paymentMethodDefaults.ts', 'utf8').match(/CARD_COLORS: string\[\] = \[([\s\S]*?)\]/)![1].match(/#[0-9A-Fa-f]{6}/g)!
  it('white >= 4.5 on the base colour of every swatch', () => {
    const bad = colours.filter((c) => ratio(WHITE, hex(c)) < 4.5)
    expect(bad, `${bad.length}/${colours.length} swatches`).toEqual([])
  })
})
